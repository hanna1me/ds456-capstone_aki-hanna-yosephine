Bike duration helper

get_bike_duration <- function(origin_lat, origin_lon,
                              dest_lat, dest_lon,
                              api_key = Sys.getenv("GOOGLE_MAPS_KEY"),
                              debug = FALSE) {

  api_key <- trimws(api_key)
  if (api_key == "") {
    return(tibble(
      bike_duration_min = NA_real_,
      bike_status       = "NO_API_KEY",
      bike_error        = "GOOGLE_MAPS_KEY not set"
    ))
  }

  coords <- c(origin_lat, origin_lon, dest_lat, dest_lon)
  if (any(!is.finite(coords))) {
    return(tibble(
      bike_duration_min = NA_real_,
      bike_status       = "INVALID_COORDS",
      bike_error        = paste("Non-finite coordinates:",
                                paste(coords, collapse = ", "))
    ))
  }

  base_url <- "https://maps.googleapis.com/maps/api/directions/json"

  query <- list(
    origin      = sprintf("%.6f,%.6f", origin_lat, origin_lon),
    destination = sprintf("%.6f,%.6f", dest_lat,  dest_lon),
    mode        = "bicycling",
    key         = api_key
  )

  full_url <- httr::modify_url(base_url, query = query)

  if (debug) {
    message("BIKE URL: ", full_url)
  }

  res <- httr::GET(full_url)

  if (httr::http_error(res)) {
    return(tibble(
      bike_duration_min = NA_real_,
      bike_status       = "HTTP_ERROR",
      bike_error        = paste0("HTTP ", httr::status_code(res))
    ))
  }

  txt <- httr::content(res, as = "text", encoding = "UTF-8")
  j   <- jsonlite::fromJSON(txt, flatten = TRUE)

  status <- j$status %||% NA_character_

  if (!identical(status, "OK") || is.null(j$routes) || nrow(j$routes) == 0) {
    return(tibble(
      bike_duration_min = NA_real_,
      bike_status       = status,
      bike_error        = if (!is.null(j$error_message)) j$error_message else NA_character_
    ))
  }

  legs_df <- j$routes$legs[[1]]

  if (is.null(legs_df) || nrow(legs_df) == 0 || is.null(legs_df$duration.value)) {
    return(tibble(
      bike_duration_min = NA_real_,
      bike_status       = status,
      bike_error        = "No duration.value in legs"
    ))
  }

  dur_sec <- legs_df$duration.value[1]

  tibble(
    bike_duration_min = dur_sec / 60,
    bike_status       = status,
    bike_error        = NA_character_
  )
}

Transit duration helper

get_transit_duration <- function(origin_lat, origin_lon,
                                 dest_lat, dest_lon,
                                 departure_time = NULL,
                                 api_key = Sys.getenv("GOOGLE_MAPS_KEY"),
                                 debug = FALSE) {

  api_key <- trimws(api_key)
  if (api_key == "") {
    return(tibble(
      transit_duration_min = NA_real_,
      transit_status       = "NO_API_KEY",
      transit_error        = "GOOGLE_MAPS_KEY not set"
    ))
  }

  coords <- as.numeric(c(origin_lat, origin_lon, dest_lat, dest_lon))
  if (any(!is.finite(coords))) {
    return(tibble(
      transit_duration_min = NA_real_,
      transit_status       = "INVALID_COORDS",
      transit_error        = paste("Non-finite coordinates:",
                                   paste(coords, collapse = ", "))
    ))
  }

  base_url <- "https://maps.googleapis.com/maps/api/directions/json"

  params <- list(
    origin      = sprintf("%.6f,%.6f", coords[1], coords[2]),
    destination = sprintf("%.6f,%.6f", coords[3], coords[4]),
    mode        = "transit",
    key         = api_key
  )

  if (!is.null(departure_time)) {
    params$departure_time <- as.integer(departure_time)
  }

 run_call <- function(params) {
    full_url <- httr::modify_url(base_url, query = params)
    if (debug) message("TRANSIT URL: ", full_url)

    res <- httr::GET(full_url)
    if (httr::http_error(res)) {
      return(list(
        status = "HTTP_ERROR",
        error  = paste0("HTTP ", httr::status_code(res)),
        j      = NULL
      ))
    }
    txt <- httr::content(res, as = "text", encoding = "UTF-8")
    j   <- jsonlite::fromJSON(txt, flatten = TRUE)
    list(status = j$status %||% NA_character_, error = j$error_message %||% NA_character_, j = j)
  }

  call1 <- run_call(params)

  if (identical(call1$status, "ZERO_RESULTS")) {
    params2 <- params
    params2$departure_time <- NULL
    call2 <- run_call(params2)
    status <- call2$status
    error  <- call2$error
    j      <- call2$j
  } else {
    status <- call1$status
    error  <- call1$error
    j      <- call1$j
  }

  if (!identical(status, "OK") || is.null(j$routes) || nrow(j$routes) == 0) {
    return(tibble(
      transit_duration_min = NA_real_,
      transit_status       = status,
      transit_error        = error
    ))
  }

  legs_df <- j$routes$legs[[1]]
  if (is.null(legs_df) || nrow(legs_df) == 0 || is.null(legs_df$duration.value)) {
    return(tibble(
      transit_duration_min = NA_real_,
      transit_status       = status,
      transit_error        = "No duration.value in legs"
    ))
  }

  dur_sec <- legs_df$duration.value[1]

  tibble(
    transit_duration_min = dur_sec / 60,
    transit_status       = status,
    transit_error        = NA_character_
  )
}
